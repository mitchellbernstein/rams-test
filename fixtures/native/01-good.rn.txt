import React, { useState } from "react";
import { Pressable, ScrollView, StyleSheet, Text, TextInput, View } from "react-native";
import { Icon } from "./Icon";

type Props = { onClose: () => void; onSignOut: () => void };

export default function ProfileSettings({ onClose, onSignOut }: Props) {
  const [name, setName] = useState("Dana Whitfield");
  const [darkMode, setDarkMode] = useState(false);

  return (
    <ScrollView contentContainerStyle={styles.screen}>
      <View style={styles.header}>
        <Pressable onPress={onClose} accessibilityRole="button" accessibilityLabel="Close" style={styles.iconButton}>
          <Icon name="x" size={24} />
        </Pressable>
        <Text accessibilityRole="header" style={styles.title}>Profile</Text>
      </View>

      <Text nativeID="nameLabel" style={styles.label}>Display name</Text>
      <TextInput
        value={name}
        onChangeText={setName}
        accessibilityLabel="Display name"
        placeholder="Your name"
        style={styles.input}
      />
      <Text style={styles.body}>Your display name appears on comments and on projects you share.</Text>

      <Pressable
        onPress={() => setDarkMode((on) => !on)}
        accessibilityRole="switch"
        accessibilityState={{ checked: darkMode }}
        style={styles.row}
      >
        <Text style={styles.rowLabel}>Dark mode</Text>
        <Text style={styles.rowValue}>{darkMode ? "On" : "Off"}</Text>
      </Pressable>

      <Pressable onPress={onSignOut} accessibilityRole="button" style={styles.row}>
        <Text style={styles.danger}>Sign out</Text>
      </Pressable>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  screen: { padding: 16, gap: 12 },
  header: { flexDirection: "row", alignItems: "center" },
  iconButton: { width: 44, height: 44, alignItems: "center", justifyContent: "center" },
  title: { fontSize: 22, fontWeight: "600", marginStart: 8 },
  label: { fontSize: 15, fontWeight: "500" },
  input: { minHeight: 44, borderWidth: 1, borderColor: "#d4d4d4", borderRadius: 8, paddingHorizontal: 12, fontSize: 16 },
  body: { fontSize: 15, color: "#555555" },
  row: { minHeight: 44, flexDirection: "row", alignItems: "center", justifyContent: "space-between", paddingVertical: 8 },
  rowLabel: { fontSize: 17 },
  rowValue: { fontSize: 17, color: "#555555", paddingEnd: 4 },
  danger: { fontSize: 17, color: "#b91c1c" },
});
